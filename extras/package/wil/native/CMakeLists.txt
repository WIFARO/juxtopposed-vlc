cmake_minimum_required(VERSION 3.21)
project(WilNativeRegression LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.8.3 EXACT REQUIRED COMPONENTS Core Gui Test)
get_filename_component(VLC_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
set(VLC_QT "${VLC_ROOT}/modules/gui/qt")
add_executable(test_dcomp_detached_pip "${VLC_QT}/tests/test_dcomp_detached_pip.cpp")
target_include_directories(test_dcomp_detached_pip PRIVATE "${VLC_QT}")
target_link_libraries(test_dcomp_detached_pip PRIVATE
    Qt6::Gui Qt6::Test d3d11 dxgi dcomp ole32 user32)
enable_testing()
add_test(NAME dcomp_detached_transfer COMMAND test_dcomp_detached_pip)
set_tests_properties(dcomp_detached_transfer PROPERTIES TIMEOUT 90)
